Bir martalik sozlash (Supabase → SQL Editor)
-- FOT Rezerv: bir martalik sozlash (qayta ishga tushirsa ham xavfsiz)
create table if not exists public.treasury_state (
  id   text primary key,
  data jsonb not null default '{}'::jsonb
);

grant select, insert, update, delete on public.treasury_state to anon, authenticated;

alter table public.treasury_state enable row level security;
drop policy if exists "fot_rezerv_rw" on public.treasury_state;
create policy "fot_rezerv_rw" on public.treasury_state
  for all to anon, authenticated using (true) with check (true);

do $$
begin
  if not exists (
    select 1 from pg_publication_tables
    where pubname = 'supabase_realtime'
      and schemaname = 'public' and tablename = 'treasury_state'
  ) then
    alter publication supabase_realtime add table public.treasury_state;
  end if;
end $$;

select 'Tayyor ✓' as holat, count(*) as qatorlar from public.treasury_state;
